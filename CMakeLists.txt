cmake_minimum_required(VERSION 3.20)
project(ZyonRobloxESP LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Este projeto requer Windows.")
endif()
add_executable(ZyonRobloxESP WIN32 src/main.cpp)
target_compile_features(ZyonRobloxESP PRIVATE cxx_std_17)
target_compile_definitions(ZyonRobloxESP PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ZyonRobloxESP PRIVATE user32 gdi32)
